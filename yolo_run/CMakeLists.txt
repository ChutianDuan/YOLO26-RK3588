cmake_minimum_required(VERSION 3.20)
project(yolo_run VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    message(FATAL_ERROR "This deployment targets native RK3588 ARM64 Linux")
endif()
find_package(Rockchip REQUIRED CONFIG PATHS /opt/rockchip/cmake NO_DEFAULT_PATH)
find_package(OpenCV REQUIRED CONFIG COMPONENTS core imgproc imgcodecs video videoio
    PATHS /usr/lib/aarch64-linux-gnu/cmake/opencv4 NO_DEFAULT_PATH)
find_package(nlohmann_json REQUIRED CONFIG PATHS /usr/share/cmake/nlohmann_json NO_DEFAULT_PATH)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
set(PKG_CONFIG_EXECUTABLE /usr/bin/pkg-config)
set(ENV{PKG_CONFIG_LIBDIR} /usr/lib/aarch64-linux-gnu/pkgconfig:/usr/lib/pkgconfig:/usr/share/pkgconfig)
set(ENV{PKG_CONFIG_PATH} "")
pkg_check_modules(SYSTEM_AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
set(YOLO_CORE_OPENCV_LIBS ${OpenCV_LIBS})
find_package(OpenCV REQUIRED CONFIG COMPONENTS highgui
    PATHS /usr/lib/aarch64-linux-gnu/cmake/opencv4 NO_DEFAULT_PATH)
add_library(yolo_core
    config/app_config.cpp image/dma_buffer.cpp image/image_processing.cpp model/yolo_engine.cpp model/yolo_postprocess.cpp
    tracking/byte_tracker.cpp video/optical_flow_tracker.cpp video/frame_stabilizer.cpp
    video/video_inference.cpp video/video_inference_detail.cpp io/result_writer.cpp)
target_include_directories(yolo_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(yolo_core PUBLIC Rockchip::RKNN Rockchip::RGA ${YOLO_CORE_OPENCV_LIBS} nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(yolo_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(yolo_run main.cpp)
target_link_libraries(yolo_run PRIVATE yolo_core)
target_compile_options(yolo_run PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(yolo_run PRIVATE YOLO_DEFAULT_CONFIG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/config.yaml")
add_library(yolo_camera camera/v4l2_camera.cpp)
target_link_libraries(yolo_camera PUBLIC yolo_core)
target_compile_options(yolo_camera PRIVATE -Wall -Wextra -Wpedantic)
add_executable(camera_run camera/main.cpp)
target_link_libraries(camera_run PRIVATE yolo_camera)
target_compile_options(camera_run PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(camera_run PRIVATE YOLO_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
# MPP import is a shared image-buffer adapter, compiled only for MPP media users.
add_library(yolo_live_media camera/live_media.cpp image/mpp_dma_buffer.cpp)
target_link_libraries(yolo_live_media PUBLIC yolo_core Rockchip::MPP)
target_compile_options(yolo_live_media PRIVATE -Wall -Wextra -Wpedantic)
add_library(yolo_live camera/live_pipeline.cpp camera/live_cli.cpp)
add_library(yolo_stream camera/stream_output.cpp)
target_link_libraries(yolo_stream PUBLIC yolo_live_media PRIVATE PkgConfig::SYSTEM_AV)
target_compile_options(yolo_stream PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(yolo_live PUBLIC yolo_stream PRIVATE opencv_highgui)
target_compile_options(yolo_live PRIVATE -Wall -Wextra -Wpedantic)
add_executable(camera_live camera/live_main.cpp)
target_link_libraries(camera_live PRIVATE yolo_live)
target_compile_definitions(yolo_live PRIVATE YOLO_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")

add_library(yolo_multi_video video/multi_video.cpp video/multi_cli.cpp video/mjpeg_recorder.cpp)
target_link_libraries(yolo_multi_video PUBLIC yolo_core PRIVATE opencv_highgui PkgConfig::SYSTEM_AV)
target_compile_options(yolo_multi_video PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(yolo_multi_video PRIVATE YOLO_MULTI_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/video_multi.yaml")
add_executable(yolo_multi video/multi_main.cpp)
target_link_libraries(yolo_multi PRIVATE yolo_multi_video)
